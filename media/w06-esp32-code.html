W06 · MC5 DATA LOGGER

โค้ด ESP32 ส่งข้อมูลเข้า Google Sheets

ESP32 อ่านค่าเซนเซอร์แล้วส่งไปที่เว็บแอปในคาบพร้อมรหัสอุปกรณ์ของกลุ่ม ค่าทุกครั้งถูกบันทึกพร้อมเวลา และขึ้นกราฟในเว็บแอป เมนู Data Logger ของกลุ่ม

ขั้นตอน

  1. ในเว็บแอป เมนู Data Logger ของกลุ่ม: Data Analyst ตั้งชื่อตัวแปร หน่วย และรอบทดสอบ แล้วกด “เปิดลิงก์ทดสอบ” ต้องเห็น "ok":true
  2. ติดตั้ง Arduino IDE และบอร์ด ESP32 (Boards Manager ค้นหา esp32 ของ Espressif Systems)
  3. กด “คัดลอกโค้ด” ด้านล่าง วางใน Arduino IDE แก้ WIFI_SSID WIFI_PASS DEVICE_KEY และฟังก์ชัน readSensors()
  4. Tools > Board: ESP32 Dev Module · Port: COM ของบอร์ด · กด Upload
  5. เปิด Serial Monitor ที่ 115200 ต้องเห็น HTTP 200 และ "ok":true แล้วดูกราฟในเว็บแอป
ความปลอดภัย ขา GPIO ของ ESP32 รับได้ไม่เกิน 3.3 V ห้ามต่อแบตเตอรี่ แผงโซลาร์ หรือไฟ 5 V เข้าขาโดยตรง ใช้วงจรแบ่งแรงดัน · ถอดสาย USB ก่อนแก้วงจร · ปั๊มและมอเตอร์ใช้แหล่งจ่ายแยก

โค้ดตัวอย่าง

/*
  IS2 THE PITCH · MC5 Data Logger
  ESP32 -> Wi-Fi -> Apps Script (เว็บแอปในคาบ) -> Google Sheets (แท็บ Logger) -> กราฟในเว็บแอป

  ขั้นตอน
  1) แก้ 3 บรรทัดในส่วน "ตั้งค่า" (Wi-Fi, รหัส Wi-Fi, รหัสอุปกรณ์ของกลุ่ม)
  2) แก้ฟังก์ชัน readSensors() ให้อ่านเซนเซอร์ของกลุ่ม ชื่อตัวแปรต้องตรงกับที่ตั้งในเว็บแอป (ไม่เกิน 4 ค่า)
  3) Tools > Board: ESP32 Dev Module · Port: COM ของบอร์ด · กด Upload
  4) เปิด Serial Monitor ที่ 115200 ต้องเห็น HTTP 200 และ "ok":true แล้วดูกราฟในเว็บแอป > Data Logger ของกลุ่ม

  ข้อควรระวัง: ESP32 ใช้ Wi-Fi 2.4 GHz เท่านั้น · Wi-Fi ที่ต้องล็อกอินผ่านหน้าเว็บใช้ไม่ได้ ให้ใช้ฮอตสปอตมือถือ
               ขา GPIO รับแรงดันได้ไม่เกิน 3.3 V ห้ามต่อแบตเตอรี่ แผงโซลาร์ หรือ 5 V เข้าขาโดยตรง ใช้วงจรแบ่งแรงดัน
*/
#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <HTTPClient.h>

// ===== ตั้งค่า =====
const char* WIFI_SSID  = "ชื่อ Wi-Fi";
const char* WIFI_PASS  = "รหัส Wi-Fi";
const char* DEVICE_KEY = "XXXXXX";   // รหัสอุปกรณ์ 6 ตัวของกลุ่ม ดูในเว็บแอป > Data Logger ของกลุ่ม
const char* EXEC_URL   = "https://script.google.com/macros/s/AKfycby9Emg9HGEli_fzRICZ2437bpW13ij-Iep_-lOuiUInV76zyv9K1-DhwPVNrQ4e00OE/exec";
const unsigned long SEND_EVERY_MS = 10000;   // ส่งทุก 10 วินาที (ทดสอบในคาบ) · เก็บข้อมูลยาวหลายชั่วโมงให้ใช้ 60000

String query;   // ค่าที่จะส่งในรอบนี้
uint32_t bootId = 0;            // สุ่มทุกครั้งที่เปิดเครื่อง ใช้กับ seq กันบันทึกซ้ำ
unsigned long sendCount = 0;

// เพิ่มค่า 1 ตัวแปร: ชื่อภาษาอังกฤษ ไม่มีเว้นวรรค ตรงกับในเว็บแอป
void addValue(const char* name, float value, int digits = 2) {
  if (isnan(value)) return;   // อ่านเซนเซอร์ไม่ได้ ไม่ส่งค่านี้
  query += "&";
  query += name;
  query += "=";
  query += String(value, digits);
}

// ===== อ่านเซนเซอร์ของกลุ่ม (แก้ตรงนี้) =====
const int ANALOG_PIN = 34;     // ตัวอย่างที่ 1: เซนเซอร์แบบแอนะล็อก เช่น ความชื้นดิน แสง (ค่า 0–4095)
const int TRIG_PIN = 5;        // ตัวอย่างที่ 2: HC-SR04 วัดระยะ (ไฟเลี้ยง 5 V ต้องแบ่งแรงดันขา ECHO ก่อนเข้า ESP32)
const int ECHO_PIN = 18;

float readDistanceCm() {
  digitalWrite(TRIG_PIN, LOW);  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  unsigned long us = pulseIn(ECHO_PIN, HIGH, 30000);   // รอไม่เกิน 30 ms
  if (us == 0) return NAN;
  return us * 0.0343 / 2.0;
}

void readSensors() {
  addValue("raw34", analogRead(ANALOG_PIN), 0);
  addValue("distance", readDistanceCm(), 1);
  // ตัวอย่างอื่น ลบ // หน้าบรรทัดเมื่อใช้ (เพิ่ม #include และตั้งค่าเซนเซอร์ด้านบนก่อน)
  // addValue("temp", dht.readTemperature(), 1);       // DHT22 ต้องติดตั้งไลบรารี DHT sensor library
  // addValue("voltage", analogReadMilliVolts(35) * 11 / 1000.0, 2);   // แบ่งแรงดัน 100k : 10k (คูณ 11)
}

// ===== ส่วนที่ไม่ต้องแก้ =====
void connectWiFi() {
  if (WiFi.status() == WL_CONNECTED) return;
  Serial.print("Wi-Fi ");
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  for (int i = 0; i < 40 && WiFi.status() != WL_CONNECTED; i++) {
    delay(500);
    Serial.print(".");
  }
  if (WiFi.status() == WL_CONNECTED) Serial.println(" OK " + WiFi.localIP().toString());
  else Serial.println(" FAIL (check SSID/password, 2.4 GHz)");
}

void sendData() {
  query = "";
  readSensors();
  if (query.length() == 0) {
    Serial.println("no sensor value");
    return;
  }
  sendCount++;
  String url = String(EXEC_URL) + "?dev=" + DEVICE_KEY + "&src=esp32&seq=" + String(bootId, HEX) + "-" + String(sendCount) + query;
  Serial.println(url);
  WiFiClientSecure client;
  client.setInsecure();                                  // ไม่ตรวจใบรับรอง (ใช้ในห้องเรียนเท่านั้น)
  HTTPClient http;
  http.setFollowRedirects(HTTPC_STRICT_FOLLOW_REDIRECTS); // Apps Script ตอบกลับด้วยการ redirect
  http.setTimeout(15000);
  if (!http.begin(client, url)) {
    Serial.println("http.begin failed");
    return;
  }
  int code = http.GET();
  if (code > 0) Serial.printf("HTTP %d %s\n", code, http.getString().c_str());
  else Serial.printf("HTTP error %s\n", http.errorToString(code).c_str());
  http.end();
}

unsigned long lastSend = 0;
bool first = true;

void setup() {
  Serial.begin(115200);
  delay(500);
  bootId = esp_random();
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  connectWiFi();
}

void loop() {
  if (first || millis() - lastSend >= SEND_EVERY_MS) {
    first = false;
    lastSend = millis();
    connectWiFi();
    if (WiFi.status() == WL_CONNECTED) sendData();
  }
}

ติดปัญหา

Serial Monitor แสดงแก้อย่างไร
Wi-Fi ...... FAILชื่อหรือรหัส Wi-Fi ผิด · ESP32 ใช้ได้เฉพาะ 2.4 GHz · Wi-Fi ที่ต้องล็อกอินผ่านหน้าเว็บใช้ไม่ได้ ให้ใช้ฮอตสปอตมือถือ
HTTP errorอินเทอร์เน็ตหลุด หรือ EXEC_URL ไม่ครบ คัดลอกลิงก์จากหน้า Data Logger ใหม่
"ok":false รหัสอุปกรณ์ไม่ถูกต้องDEVICE_KEY ไม่ตรงกับหน้า Data Logger ของกลุ่ม
"ok":false ส่งถี่เกินไปเพิ่ม SEND_EVERY_MS เป็นอย่างน้อย 10000
"ok":false ไม่มีค่าตัวเลขเซนเซอร์อ่านไม่ได้ทุกค่า (NAN) ตรวจสายและไลบรารี
ค่าเป็น 0 หรือ 4095 ตลอดต่อผิดขา · ใช้ขาแอนะล็อก 32–39 (ขากลุ่ม ADC2 ใช้ไม่ได้ตอนเปิด Wi-Fi)
Upload ไม่ได้เลือก Board/Port ผิด · สาย USB ชาร์จอย่างเดียว · กดปุ่ม BOOT ค้างตอนเริ่ม Upload
